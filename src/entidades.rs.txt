use crate::Tecla;
use std::io::{stdout, Write};
const ANCHO_PANTALLA: u32 = 80;
const ALTO_PANTALLA: u32 = 24;
#[derive(Clone, Copy)]
struct Posicion {
    x: usize,
    y: usize,
}
#[derive(Clone, Copy)]
struct Disparo {
    pos: Posicion,
    dir: i32,
}
#[derive(Clone, Copy)]
pub struct Enemigo {
    pos: Posicion,
    vivo: bool,
}
#[derive(Clone, Copy)]
struct VectorDisparo {
    dimension: usize,
    elementos: [Disparo; 20],
}
impl VectorDisparo {
    fn new () -> Self {
        VectorDisparo {
            dimension: 0,
            elementos: [Disparo { pos: Posicion { x: 0, y: 0 }, dir: 0 }; 20],
        }
    }
    fn dim (&self) -> usize {
        self.dimension
    }
    fn add (&mut self, disparo: Disparo) {
        if self.dimension < 20 {
            self.elementos[self.dimension] = disparo;
            self.dimension += 1;
        }
    }
    fn actualizar(&mut self) {
        let mut i = 0;
        while i < self.dimension {
            let mut disparo = self.elementos[i];
            disparo.pos.y = (disparo.pos.y as i32 + disparo.dir) as usize;
            if disparo.pos.y >= ALTO_PANTALLA as usize || disparo.pos.y == 0 {
                for j in i..self.dimension.saturating_sub(1) {
                    self.elementos[j] = self.elementos[j + 1];
                }
                self.dimension -= 1;
            } else {
                self.elementos[i] = disparo;
                i += 1;
            }
        }
    }
}
struct Jugador {
    pos: Posicion,
    vida: u32,
}
impl Jugador {
    fn new() -> Self {
        Jugador {
            pos: Posicion { x: ANCHO_PANTALLA as usize / 2, y: ALTO_PANTALLA as usize - 1 },
            vida: 3,
        }
    }
    fn mover(&mut self, tecla: crate::input::Tecla) {
    match tecla {
        crate::input::Tecla::FlechaIzquierda => {
            if self.pos.x > 0 {
                self.pos.x -= 1;
            }
        }
        crate::input::Tecla::FlechaDerecha => {
            if self.pos.x < ANCHO_PANTALLA as usize - 1 {
                self.pos.x += 1;
            }
        }
        _ => {}
    }    
    }
}
pub struct Juego {
    jugador: Jugador,
    enemigos: [Enemigo; 20],
    disparos: VectorDisparo,
    num_enemigos: usize,
    dir_enemigos: i32,
    puntos: u32,
}
impl Juego {
    pub fn new() -> Self{
        let mut juego = Self {
            jugador: Jugador::new(),
            enemigos: [Enemigo { pos: Posicion { x: 0, y: 0 }, vivo: false }; 20],
            disparos: VectorDisparo::new(),
            num_enemigos: 0,
            dir_enemigos: 1,
            puntos: 0,
        };
        juego.crear_enemigos();
        juego
    }

    fn crear_enemigos(&mut self){
        let inicio_x = 10;
        let espaciado_x = 4;
        for i in 0..8{
            self.enemigos[self.num_enemigos] = Enemigo { pos: Posicion { x: inicio_x + i * espaciado_x, y: 3 }, vivo: true };
            self.num_enemigos += 1;
        }
        for i in 0..8{
            self.enemigos[self.num_enemigos]= Enemigo { pos: Posicion { x: inicio_x + i * espaciado_x, y: 6 }, vivo: true };
            self.num_enemigos += 1;
        }
    }

    pub fn actualizar(&mut self){
        print!("\x1B[2J\x1B[H");
        print!("Vidas: {} Puntos: {:04}    \r", self.jugador.vida, self.puntos);
        // Jugador
        print!("\x1B[{};{}H▲", self.jugador.pos.y + 1, self.jugador.pos.x + 1);
        // Disparos
        self.disparos.actualizar();
        for i in 0..self.disparos.dim(){
            let d = self.disparos.elementos[i];
            if d.pos.y < ALTO_PANTALLA as usize && d.pos.y > 0 {
                print!("\x1B[{};{}H|", d.pos.y + 1, d.pos.x + 1);
            }
        }
        // Enemigos
        self.mover_enemigos();
        for i in 0..self.num_enemigos {
            if self.enemigos[i].vivo {
                print!("\x1B[{};{}H■", self.enemigos[i].pos.y + 1, self.enemigos[i].pos.x + 1);
            }
        }
        let _ = stdout().flush();
    }

    pub fn dibujar(&self){
        // actualizar() ya se encarga de dibujar, así que aquí no hacemos nada
    }

    pub fn mover_enemigos(&mut self){
        let mut borde = false;
        for i in 0..self.num_enemigos{
            if self.enemigos[i].vivo {
                let nueva_x = (self.enemigos[i].pos.x as i32 + self.dir_enemigos) as usize;
                if nueva_x == 0 || nueva_x == ANCHO_PANTALLA as usize - 2 {
                    borde = true;
                    break;
                }
            }
        }
        if borde {
            self.dir_enemigos *= -1;
            for i in 0..self.num_enemigos {
                if self.enemigos[i].vivo {
                    self.enemigos[i].pos.y += 1;
                }
            }
        } else {
            for i in 0..self.num_enemigos {
                if self.enemigos[i].vivo {
                    self.enemigos[i].pos.x = (self.enemigos[i].pos.x as i32 + self.dir_enemigos) as usize;
                }
            }
        }
    }

    pub fn procesar_tecla(&mut self, tecla: Tecla){
        match tecla {
            Tecla::FlechaIzquierda | Tecla::FlechaDerecha => {
                self.jugador.mover(tecla);
            }
            Tecla::Espacio => {
                self.disparos.add(Disparo { pos: Posicion { x: self.jugador.pos.x, y: self.jugador.pos.y - 1 }, dir: -1 });
            }
            Tecla::Q => {
                crossterm::terminal::disable_raw_mode().ok();
                println!("\n¡Gracias por jugar!\n");
                std::process::exit(0);
            }
            _ => {}
        }
    }
}
